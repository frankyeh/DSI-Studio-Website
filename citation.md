## General citation for DSI Studio and Fiber Data Hub

When citing DSI Studio, cite the 2025 *Nature Methods* paper and report the **exact DSI Studio release used for the analysis**. The version family (Sun, Hou, Chen, etc.) may be included as additional information, but it should not replace the exact release date.

<div class="cite-statement"><p class="cite-eyebrow">Example methods statement</p><p class="cite-text" id="cite-methods">The analysis was conducted using DSI Studio (Yeh, 2025), version YYYY.MM.DD (Sun).</p><div class="cite-actions"><button type="button" class="cite-copy" data-copy="cite-methods" hidden><span class="cite-copy-label">Copy statement</span></button></div></div>

For reproducibility, keep a copy of the DSI Studio version used for each project and report relevant reconstruction/tracking parameters in the methods.

(function () {
  var status = document.querySelector(".cite-status");
  function copied(btn, ok) {
    var label = btn.querySelector(".cite-copy-label"), text = label.dataset.text || label.textContent;
    label.dataset.text = text;
    label.textContent = ok ? "Copied" : "Press Ctrl+C";
    btn.classList.toggle("is-done", ok);
    if (status) status.textContent = ok ? "Copied to clipboard." : "Copy failed; the text is selected for manual copying.";
    setTimeout(function () { label.textContent = text; btn.classList.remove("is-done"); }, 1800);
  }
  function select(el) {
    var r = document.createRange(); r.selectNodeContents(el);
    var s = window.getSelection(); s.removeAllRanges(); s.addRange(r);
  }
  document.querySelectorAll(".cite-copy").forEach(function (btn) {
    var el = document.getElementById(btn.dataset.copy);
    if (!el) return;
    btn.hidden = false;
    btn.addEventListener("click", function () {
      var text = el.textContent.trim();
      if (navigator.clipboard && window.isSecureContext) {
        navigator.clipboard.writeText(text).then(function () { copied(btn, true); },
          function () { select(el); copied(btn, false); });
      } else {
        select(el);
        var ok = false;
        try { ok = document.execCommand("copy"); } catch (e) {}
        copied(btn, ok);
      }
    });
  });
})();
</script>
